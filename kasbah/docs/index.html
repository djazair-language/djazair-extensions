Kasbah Web Framework for Djazair

An enterprise-grade, modular, high-performance, and secure web framework built natively for the Djazair programming language. Designed around modern global standards (Express, Fastify, Laravel, FastAPI, Spring).

100% Test Pass Rate (235 Tests) CORS & Helmet Protected 100% Pure Djazair
Terminal 
dpm install kasbah

Overview & Architecture

Kasbah is engineered to deliver a world-class developer experience in Djazair without sacrificing speed, security, or readability:

Standalone Modular Routers

Separate routing logic into dedicated router files and mount them cleanly with app.mount("/prefix", subRouter).

Intrinsic Multi-Format Body Parser

Automatic parsing for JSON, URL-encoded forms, and multipart uploads without requiring manual middleware registration.

Secure by Default (CORS & Helmet)

Built-in protections against HTTP Response Splitting, Clickjacking, MIME-sniffing, XSS, and full Cross-Origin resource sharing.

Standardized Fluent APIs

Predictable predicate checkers (isJson, isAjax, isSecure, isType) and chainable response builders.

Quick Start

Create a file named server.dz and launch your server in just a few lines of code:

server.dz 
use kasbah

# 1. Initialize application
let app = new kasbah.app({
    "port": 8080,
    "static": "public"
})

# 2. Register routes
app.get("/", fn(req, res)
    res.html("<h1>Welcome to Kasbah Web Framework!</h1>")
end)

app.get("/api/hello/:name", fn(req, res)
    res.json({
        "message": "Hello, ${req.param(\"name\")}!",
        "isAjax": req.isAjax(),
        "version": "0.2.0"
    })
end)

# 3. Start server
app.listen()

Run your server via terminal:

Terminal 
djazair server.dz

Standard Configuration Schema

Kasbah applications accept a clean configuration map supporting flat options and structured nested objects:

app_config.dz 
let app = new kasbah.app({
    # ── Core Options ──────────────────────────────────────────
    "port":        8080,          # TCP port to listen on (default: 3000)
    "host":        "0.0.0.0",     # Binding host address (default: "0.0.0.0")
    "static":      "public",      # Directory path for static assets (CSS, JS, images)
    "views":       "views",       # Directory path for Qalam template views
    "maxBodySize": 10,            # Maximum request payload size in MB (default: 10)
    "debug":       False,         # Expose error details in development mode
    "logger":      True,          # Request/error console logging
    "trustProxy":  False,         # Trust X-Forwarded-For reverse proxy headers

    # ── Session Management (Nested Object or One-liner) ───────
    "session": {
        "secret":   "my-secret-key-at-least-32-chars-long-secure", # (Required) >= 32 chars
        "maxAge":   86400,        # Cookie lifetime in seconds (default: 1 day)
        "name":     "kasbah.sid", # Cookie name
        "secure":   False,        # HTTPS only flag
        "sameSite": "Lax"         # SameSite policy: "Lax", "Strict", "None"
    },

    # ── Advanced Server / HTTP Settings (Optional) ────────────
    "server": {
        "keepAlive":             False,
        "keepAliveTimeout":      30,   # Keep-alive idle timeout in seconds
        "maxKeepAliveRequests":  100   # Max requests per persistent connection
    }
})

Advanced Routing System

Kasbah comes with a dedicated radix-style routing engine handling dynamic parameters, wildcard grouping, and modular sub-routers:

routes.dz 
# Standard HTTP Methods
app.get("/trades", handler)
app.post("/trades", handler)
app.put("/trades/:id", handler)
app.delete("/trades/:id", handler)
app.patch("/trades/:id", handler)
app.all("/webhook", handler) # Matches all HTTP verbs

# Route-Level Middleware (Single or Array)
app.get("/dashboard", authGuard, fn(req, res)
    res.ok({"user": req.get("user")})
end)

app.post("/admin/settings", [authGuard, adminOnly], fn(req, res)
    res.ok({"status": "saved"})
end)

# Multiple dynamic route parameters
app.get("/users/:userId/trades/:tradeId", fn(req, res)
    let uid = req.param("userId")
    let tid = req.param("tradeId")
    res.json({"user": uid, "trade": tid})
end)

Standalone Routers & Mounting

Isolate routes into modular files with kasbah.router, and mount them in your application with app.mount(prefix, subRouter):

routes/trades.dz 
import "../kasbah/init.dz" as kasbah

let tradesRouter = new kasbah.router()

tradesRouter.get("/", fn(req, res)
    res.ok({"trades": []})
end)

tradesRouter.get("/:id", fn(req, res)
    res.ok({"id": req.param("id")})
end)
app.dz 
import "routes/trades.dz" as tradesMod

# Mount the sub-router under /api/trades
app.mount("/api/trades", tradesMod.tradesRouter)

Route Groups & Scoped Middleware

Organize related routes under a common URL prefix and apply scoped middleware pipelines to all routes within the group:

route_groups.dz 
# Define scoped authentication middleware
let authMiddleware = fn(req, res)
    if isNull(req.token())
        res.unauthorized("Authentication token is required")
        return
    end
end

# Register /admin route group with authMiddleware
app.group("/admin", [authMiddleware], fn(g)
    g.get("/dashboard", fn(req, res)
        res.json({"status": "admin dashboard"})
    end)

    g.get("/users", fn(req, res)
        res.json({"users": ["Alice", "Bob"]})
    end)

    # Nested sub-groups
    g.group("/settings", fn(settingsGroup)
        settingsGroup.get("/security", fn(req, res)
            res.ok({"2fa": True})
        end)
    end)
end)

Request (`request`) Object

The req instance wraps incoming HTTP data with standardized getters and boolean predicate checkers:

Method / Property Return Type Description
req.method String HTTP method (e.g. "GET", "POST").
req.path String Normalized, URL-decoded request path.
req.body Map / String Parsed body (JSON dictionary or form-urlencoded fields).
req.params Map Dictionary of route URL parameters (e.g. {"id": "42"}).
req.param(name, fallback) Any Retrieves route URL parameter (e.g. from /trades/:id).
req.query(name, fallback) Any / Map Retrieves URL query parameter (?page=1) or entire query map.
req.input(name, fallback) Any Looks up key across params → query → body.
req.all() / inputs() Map Merges params + query + body into unified map.
req.only(keys) Map Filters inputs to only the specified allowed keys.
req.except(keys) Map Returns all inputs excluding the specified keys.
req.isJson() Boolean Checks if client requests or sends JSON (Accept, Content-Type, or AJAX).
req.isAjax() Boolean Checks if request was sent with X-Requested-With: XMLHttpRequest.
req.isSecure() / secure() Boolean Checks if request was made via HTTPS or TLS termination proxy.
req.isType(cat) / contentIs(cat) Boolean Checks Content-Type ("json", "form", "multipart").
req.token() / bearerToken() String / Null Extracts Bearer token from Authorization: Bearer <token>.
req.header(name) String Retrieves HTTP header value (case-insensitive).
req.cookie(name) String / Null Retrieves cookie value from incoming request.
req.file(name) uploadedFile / Null Retrieves uploaded file descriptor for multipart forms.
req.hasFile(name) Boolean Verifies if an uploaded file exists and is saved on disk.
req.uploader(options) fileManager Creates configured file manager uploader instance.
req.validate(data) validator Creates validator instance bound to request body.
req.set(k, v) / get(k) Self / Any Store and retrieve custom middleware state across handlers.

Response (`response`) Object

The res instance provides a fluent, chainable API for crafting HTTP responses:

response_examples.dz 
# Status code and response headers
res.status(200).header("X-Powered-By", "Kasbah").json({"status": "success"})

# HTML, JSON, Text, and XML sending
res.html("<h1>Custom HTML Page</h1>")
res.text("Plain text content")
res.json({"status": "ok"})
res.xml("<root><item>1</item></root>")

# File streaming and downloads with Unicode support
res.file("public/reports/annual.pdf")
res.download("public/export.csv", "trades_report.csv")

# Cookie management and redirects
res.cookie("theme", "dark", {"maxAge": 86400, "httpOnly": True})
res.clearCookie("theme")
res.redirect("/dashboard")
res.back() # Redirect back to Referer URL

Semantic Response Helpers

Convenience one-liner helpers for returning standard HTTP status responses with clean JSON payloads:

Helper Method HTTP Status Typical Usage
res.ok(data) 200 OK Sends 200 response (serializes data as JSON if provided).
res.created(data) 201 Created Confirms successful creation of a new resource.
res.noContent() 204 No Content Empty success response (common after DELETE operations).
res.badRequest(msg) 400 Bad Request Returns error JSON {"error": msg}.
res.unauthorized(msg) 401 Unauthorized Authentication failure or expired session.
res.forbidden(msg) 403 Forbidden Access forbidden due to insufficient permissions.
res.notFound(msg) 404 Not Found Requested resource does not exist.
res.validationError(errors) 422 Unprocessable Entity Directly returns validator error dictionary {"errors": errors}.
res.serverError(msg) 500 Internal Error Internal server failure JSON response.

CORS Middleware (`kasbah.cors`)

Manage Cross-Origin Resource Sharing with full support for origins, credentials, custom headers, and automated preflight OPTIONS handling:

cors_example.dz 
use kasbah

let app = new kasbah.app()

# 1. Default (allow all origins '*')
app.middleware(kasbah.cors())

# 2. Enterprise CORS configuration with credentials
app.middleware(kasbah.cors({
    "origin":            ["https://app.mysite.com", "https://admin.mysite.com"],
    "methods":           ["GET", "POST", "PUT", "DELETE"],
    "allowedHeaders":    ["Content-Type", "Authorization", "X-Requested-With"],
    "credentials":       True,
    "maxAge":            86400,
    "preflightContinue": False
}))

Helmet Security Headers (`kasbah.helmet`)

Secure your application by automatically configuring essential HTTP security headers:

helmet_example.dz 
use kasbah

let app = new kasbah.app()

# 1. Default Security Headers
# Injects X-Frame-Options, nosniff, X-XSS-Protection, HSTS, Referrer-Policy, and hides X-Powered-By
app.middleware(kasbah.helmet())

# 2. Custom Security Headers & CSP
app.middleware(kasbah.helmet({
    "xFrameOptions":  "DENY",
    "referrerPolicy": "strict-origin-when-cross-origin",
    "hsts": {
        "maxAge":            31536000,
        "includeSubDomains": True,
        "preload":           True
    },
    "csp": {
        "default-src": ["'self'"],
        "script-src":  ["'self'", "https://cdn.jsdelivr.net"]
    }
}))

Validation Engine (`validator`)

Declarative schema validation with chained constraint rules and error mapping:

validation_example.dz 
app.post("/api/trades", fn(req, res)
    # Create validator from request body
    let v = req.validate()

    # Define validation schema rules
    v.field("symbol").required().string().minLength(2).maxLength(10)
    v.field("type").required().oneOf(["BUY", "SELL"])
    v.field("quantity").required().number().min(1)
    v.field("price").required().number().min(0.01)
    v.field("email").optional().email()

    # Validate constraints
    if v.fails()
        return res.validationError(v.errors())
    end

    # Extract sanitized, validated data
    let cleanData = v.validated()
    res.created({"message": "Trade recorded successfully", "trade": cleanData})
end)

File Uploads & FileManager

Secure file upload processing with extension whitelisting, MIME checks, and automated temporary file cleanup:

upload_handler.dz 
app.post("/upload/avatar", fn(req, res)
    # 1. Verify file presence
    if not req.hasFile("avatar")
        return res.badRequest("No avatar uploaded")
    end

    let file = req.file("avatar")

    # 2. Check extension & MIME type
    if not file.isImage() or not file.hasExt(["jpg", "png", "webp"])
        return res.badRequest("Invalid image file format")
    end

    # 3. Save to target directory
    let savedPath = file.saveTo("public/uploads/avatars")
    res.ok({"status": "success", "path": "/" + savedPath})
end)

Full REST Application Example

A complete production-ready application demonstrating routing, middleware, sessions, validation, and APIs:

app_complete.dz 
use kasbah

# Initialize Application with standard config
let app = new kasbah.app({
    "port":       8080,
    "static":     "public",
    "bodyParser": True,
    "logger":     True,
    "session": {
        "secret": "kasbah-production-secret-key-32-chars-long!",
        "maxAge": 86400
    }
})

# Global Security Middlewares
app.middleware(kasbah.helmet())
app.middleware(kasbah.cors({
    "origin":      "*",
    "credentials": True
}))

# In-Memory Store
let trades = []

# HTML Route
app.get("/", fn(req, res)
    let visits = req.session.get("visits", 0) + 1
    req.session.set("visits", visits)
    res.html("<h1>Kasbah Trades API</h1><p>Visits: " + str(visits) + "</p>")
end)

# REST API with Validation
app.post("/api/trades", fn(req, res)
    let v = req.validate()
    v.field("symbol").required().string().minLength(2)
    v.field("price").required().number().min(0.01)

    if v.fails()
        return res.validationError(v.errors())
    end

    let newTrade = {
        "id":     trades.length() + 1,
        "symbol": req.input("symbol"),
        "price":  req.input("price")
    }
    trades.append(newTrade)

    res.created({"message": "Trade recorded", "trade": newTrade})
end)

# 404 Route Handler
app.onNotFound(fn(req, res)
    if req.isJson()
        res.notFound("Requested resource not found")
    else
        res.status(404).html("<h1>404 Not Found</h1><p>Path: ${req.path}</p>")
    end
end)

# Start the Server
app.listen()
